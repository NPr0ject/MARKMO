using System;
class programm
{

    static void Main()
    {
        int m = Convert.ToInt32(Console.ReadLine()); // кол - во  строк массива
        int n = Convert.ToInt32(Console.ReadLine()); // кол - во  столбцов массива
        int[,] a = new int[m, n];
        int[] plus = new int[m];
        int[] proiz = new int[m];
        int[] noly = new int[m];
        for (int i = 0; i < m; i++)
        {
            for (int j = 0; j < n; j++)
            {
                a[i, j] = Convert.ToInt32(Console.ReadLine()); // читаем элементы в строкe
                plus[i] = plus[i] + a[i, j];
                if (a[i, j] == 0)
                {
                    noly[i]++;
                }
                else
                {
                    proiz[i] = proiz[i] * a[i, j];
                }
            }
        }
        for (int i = 0; i < m; i++)
        { // все от 0 до n
            for (int j = i + 1; j < m; j++)
            { // все следующие
                if (noly[i] == noly[j] && plus[i] == plus[j] && proiz[i] == proiz[j])
                {
                    Console.WriteLine((i + 1) + " " + (j + 1)); // для коректного вывода
                }
            }
        }
    }
}
