using System;
class programm
{
    static void Main()
    {
        int n = Convert.ToInt32(Console.ReadLine());// кол - во массивов
        int[][] a = new int[n][];
        int[] max = new int[n];
        int[] min = new int[n];
        string ans = "";
        for (int i = 0; i < n; i++)
        {
            int m = Convert.ToInt32(Console.ReadLine()); // кол-во элеметов n-ого массива
            a[i] = new int[m];
            for (int j = 0; j < m; j++)
            {
                a[i][j] = Convert.ToInt32(Console.ReadLine());
                if (a[i][j] <= min[i])
                {
                    min[i] = j;
                }
                if (a[i][j] >= max[i])
                {
                    max[i] = j;
                }
            }
            Console.WriteLine(i + 1 + "масив введён");
            if (max[i] % 2 == 0 && min[i] % 2 == 0)
            {
                ans += (i + 1);
            }
        }
            Console.WriteLine(ans);
    }
}
