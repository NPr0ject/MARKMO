using System;
class programm
{
    static void Main()
    {
        int n = Convert.ToInt32(Console.ReadLine());
        int[] a = new int[n];
        string ans = "";
        for (int i = 0; i < n; i++)
        {
            a[i] = Convert.ToInt32(Console.ReadLine());
            if (a[i] == 0)
            {
                ans = '0' + ans;
            }
            if (a[i] > 0)
            {
                ans = ans + a[i];
            }
        }
        for (int i = 0;i < n; i++)
        {
            if (a[i] < 0)
            {
                ans = a[i] + ans;
            }
        }
        Console.WriteLine(ans);
    }
}
