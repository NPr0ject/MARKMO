using System;
class programm
{
    static void Main()
    {
        int n = Convert.ToInt32(Console.ReadLine());
        int[] a = new int[n];
        bool ans = true;
        a[0] = Convert.ToInt32(Console.ReadLine());
        a[1] = Convert.ToInt32(Console.ReadLine());
        int step = a[1] - a[0];
        for (int i = 2; i < n; i++)
        {
            a[i] = Convert.ToInt32(Console.ReadLine());
            if (a[i] <= a[i - 1] || a[i] - a[i - 1] != step)
            {
                ans = false;
            }

        }
            Console.WriteLine(ans);
    }
}
