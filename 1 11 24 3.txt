using System;
class programm
{

    static void Main()
    {
        int m = Convert.ToInt32(Console.ReadLine()); // кол - во  строк массива
        int n = Convert.ToInt32(Console.ReadLine()); // кол - во  столбцов массива
        int[,] a = new int[m, n];
        int[] sum = new int[n];
        for (int i = 0; i < m; i++)
        {
            for (int j = 0; j < n; j++)
            {
                a[i, j] = Convert.ToInt32(Console.ReadLine()); // читаем элементы в строкe
                sum[i] += a[i, j];
            }
        }
        bool ans = true;
        for (int i = 0; i < m; i++)
        {
            if (sum[i]%2 != 0)
            {
                ans = false;
            }
        }
        Console.WriteLine(ans);
    }
}
