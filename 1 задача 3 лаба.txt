using System;
class programm {
    // задание 1
    static void Main() {
        int n = Convert.ToInt32(Console.ReadLine());
        int ans = 0;
        int score = 0;
    for (int i = 0; i < n; i++){
        int c = Convert.ToInt32(Console.ReadLine());
        if (c != 0){
            score++;
        }
        else {
            if(ans < score){
                ans = score;
            }
            score = 0;
        }
    }
    if(ans == 0)
        ans = score;
    Console.WriteLine(ans);
    }
}
